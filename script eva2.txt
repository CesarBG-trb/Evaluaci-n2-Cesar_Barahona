import requests
import urllib.parse

# URL base para la API de Graphhopper
url_geocodificacion = "https://graphhopper.com/api/1/geocode?"
url_ruta = "https://graphhopper.com/api/1/route?"

# === REEMPLAZA CON TU API KEY DE GRAPHHOPPER ===
clave_api = "9c6913e9-1996-4fa8-8227-d0efab344fcf"

# ============================================================
# FUNCIÓN: Geocodificación (obtener latitud y longitud)
# ============================================================
def geocodificar(ubicacion, clave):
    """Convierte una ciudad en coordenadas (lat, lng)"""
    # Si la ubicación está vacía, pedir de nuevo
    while ubicacion == "":
        ubicacion = input("Ingrese la ubicación nuevamente: ")
    
    # Construir la URL con parámetros
    url = url_geocodificacion + urllib.parse.urlencode(
        {"q": ubicacion, "limit": "1", "key": clave}
    )
    
    # Realizar la solicitud
    respuesta = requests.get(url)
    datos_json = respuesta.json()
    codigo_estado = respuesta.status_code
    
    if codigo_estado == 200 and len(datos_json["hits"]) != 0:
        lat = datos_json["hits"][0]["point"]["lat"]
        lng = datos_json["hits"][0]["point"]["lng"]
        nombre = datos_json["hits"][0]["name"]
        tipo = datos_json["hits"][0]["osm_value"]
        
        # Construir nombre completo
        if "country" in datos_json["hits"][0]:
            pais = datos_json["hits"][0]["country"]
        else:
            pais = ""
        
        if "state" in datos_json["hits"][0]:
            estado = datos_json["hits"][0]["state"]
        else:
            estado = ""
        
        if len(estado) != 0 and len(pais) != 0:
            ubicacion_completa = nombre + ", " + estado + ", " + pais
        elif len(estado) != 0:
            ubicacion_completa = nombre + ", " + pais
        else:
            ubicacion_completa = nombre
        
        print(f"URL de Geocodificación para {ubicacion_completa} (Tipo: {tipo})")
        print(url)
        print()
    else:
        lat = "null"
        lng = "null"
        ubicacion_completa = ubicacion
        if codigo_estado != 200:
            print(f"Error en Geocodificación. Código: {codigo_estado}")
            print(f"Mensaje: {datos_json.get('message', 'Error desconocido')}")
    
    return codigo_estado, lat, lng, ubicacion_completa

        break

    origen = geocodificar(ciudad_inicio, clave_api)
